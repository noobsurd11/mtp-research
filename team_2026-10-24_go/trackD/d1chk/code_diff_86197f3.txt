+"""Track D: data.labeled_list (explicit labelled stems) in scripts/train.py build_benchmark_data."""
+import sys
+from pathlib import Path
+
+import pytest
+import torch
+
+ROOT = Path(__file__).resolve().parent.parent
+sys.path.insert(0, str(ROOT)); sys.path.insert(0, str(ROOT / "scripts"))
+
+from src.data.splits import load_split
+from src.utils.config import load_config
+from train import build_benchmark_data
+
+SPLITS = ROOT / "data" / "splits"
+pytestmark = pytest.mark.skipif(not (ROOT / "data/labeled/images").is_dir(), reason="data not linked")
+
+
+def _cfg(**data):
+    cfg = load_config(ROOT / "configs" / "supervised.yaml")
+    cfg["data"].update(dict(labeled_fraction=0.05, split_seed=0, mask_subdir="labeled/masks_escdd",
+                            augmentation="labelsafe", root=str(ROOT / "data")), **data)
+    cfg["training"]["labeled_batch_size"] = 8
+    return cfg
+
+
+
+def test_list_equal_to_fraction_file_gives_same_meta(tmp_path):
+    a = build_benchmark_data(_cfg(), "supervised", 0, 0.05, False, torch.device("cpu"), 0)
+    b = build_benchmark_data(_cfg(labeled_list=str(SPLITS / "seed0" / "labeled_5pct.txt")), "supervised", 0, 0.05,
+                             False, torch.device("cpu"), 0)
+    assert (a["class_counts"] == b["class_counts"]).all()
+    ma, mb = a["info"], b["info"]
+    assert ma["n_labeled"] == mb["n_labeled"] == 23
+    assert {k: v for k, v in ma.items() if k != "labeled_split"} == {k: v for k, v in mb.items() if k != "labeled_split"}
+
+
+def test_ssl_pool_is_rest_of_train(tmp_path):
+    lab = load_split(SPLITS / "seed0" / "labeled_5pct.txt")
+    b = build_benchmark_data(_cfg(labeled_list=str(SPLITS / "seed0" / "labeled_5pct.txt")), "mean_teacher", 0, 0.05,
+                             True, torch.device("cpu"), 0)
+    assert b["info"]["n_unlabeled"] == 478 - len(lab)
+
+
+def test_rejects_non_train_and_duplicates(tmp_path):
+    bad = tmp_path / "bad.txt"
+    bad.write_text("\n".join(load_split(SPLITS / "test.txt")[:2]) + "\n")
+    with pytest.raises(ValueError, match="non-train"):
+        build_benchmark_data(_cfg(labeled_list=str(bad)), "supervised", 0, 0.05, False, torch.device("cpu"), 0)
+    dup = tmp_path / "dup.txt"
+    s = load_split(SPLITS / "seed0" / "labeled_5pct.txt")
+    dup.write_text("\n".join(s + s[:1]) + "\n")
+    with pytest.raises(ValueError, match="duplicate"):
+        build_benchmark_data(_cfg(labeled_list=str(dup)), "supervised", 0, 0.05, False, torch.device("cpu"), 0)
+
+
